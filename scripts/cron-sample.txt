# Sample cron entry to update data and rebuild daily at 2:00 AM
# Place this line into your crontab (crontab -e) or add via cPanel -> Cron Jobs
# Change /home/username/path-to-app to your installation path

# m h  dom mon dow   command
0 2 * * * /bin/bash /home/username/path-to-app/scripts/update-and-build.sh >> /home/username/path-to-app/logs/update-and-build.log 2>&1
